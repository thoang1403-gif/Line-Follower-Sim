cmake_minimum_required(VERSION 3.16)
project(LineFollowerSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets SerialPort)
add_library(line_core core/engine.cpp)
target_include_directories(line_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(LineFollowerSim main.cpp app/simulator.cpp app/controller.cpp app/source.cpp app/journal.cpp app/model_io.cpp)
target_link_libraries(LineFollowerSim PRIVATE line_core Qt6::Widgets Qt6::SerialPort)
set_target_properties(LineFollowerSim PROPERTIES WIN32_EXECUTABLE TRUE)
if(MSVC)
    target_compile_options(LineFollowerSim PRIVATE /utf-8)
    target_compile_options(line_core PRIVATE /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME gui_smoke COMMAND ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen
        $<TARGET_FILE:LineFollowerSim> --smoke-test ${CMAKE_CURRENT_BINARY_DIR}/ui-smoke)
    set_tests_properties(gui_smoke PROPERTIES TIMEOUT 30)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE line_core)
    add_test(NAME core_tests COMMAND core_tests)
    add_executable(model_check tests/model_check.cpp app/model_io.cpp)
    target_link_libraries(model_check PRIVATE line_core Qt6::Core)
    add_executable(arduino_adapter_tests tests/arduino_adapter_tests.cpp)
    target_include_directories(arduino_adapter_tests PRIVATE tests/arduino_stub arduino/TelemetryDemo)
    add_test(NAME arduino_adapter_tests COMMAND arduino_adapter_tests)
    # Host-only firmware regression test; never link Arduino code into the GUI.
    add_executable(arduino_control_tests tests/arduino_control_tests.cpp)
    target_include_directories(arduino_control_tests PRIVATE tests/arduino_stub)
    target_link_libraries(arduino_control_tests PRIVATE line_core)
    add_test(NAME arduino_control_tests COMMAND arduino_control_tests)
    add_test(NAME arduino_adapter_copies_match COMMAND ${CMAKE_COMMAND} -E compare_files
        ${CMAKE_CURRENT_SOURCE_DIR}/arduino/TelemetryDemo/TelemetryLink.h
        ${CMAKE_CURRENT_SOURCE_DIR}/arduino/Control_Hardware/TelemetryLink.h)
endif()
